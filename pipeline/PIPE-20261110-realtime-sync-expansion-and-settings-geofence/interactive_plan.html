2. ชุดคำสั่ง SQL สำหรับ staging + สรุปสะสมคำสั่ง SQL สำหรับ public

จัดทำชุดคำสั่ง SQL สำหรับรันบนสคีมา staging (sql/create_staging_schema.sql) และอัปเดตสรุปคำสั่ง SQL สะสมทั้งหมดที่ต้องนำไปรันบนสคีมา public (sql/alter_v2_8_loca_v2_schema.sql) ควบคู่กันทุกครั้งที่มีการปรับโครงสร้างตารางบน staging

1. ชุดคำสั่ง SQL สำหรับ Schema staging และสรุปคำสั่ง SQL สะสมสำหรับ Schema public

1.1 คำสั่ง SQL สำหรับรันบน staging ตอนนี้ (อัปเดตใน sql/create_staging_schema.sql)

-- ============================================================================
-- [STAGING] เพิ่มคอลัมน์ enforce_geofence_radius และเปิด Realtime Publication บน staging
-- ============================================================================
ALTER TABLE staging.system_settings
  ADD COLUMN IF NOT EXISTS enforce_geofence_radius BOOLEAN NOT NULL DEFAULT true;

UPDATE staging.system_settings
SET enforce_geofence_radius = COALESCE(
  (category_colors->>'enforce_geofence_radius')::boolean,
  enforce_geofence_radius,
  true
);

-- เปิดใช้งาน Supabase Realtime WebSocket สำหรับตารางในสคีมา staging
DO $$
BEGIN
  IF EXISTS (SELECT 1 FROM pg_publication WHERE pubname = 'supabase_realtime') THEN
    BEGIN
      ALTER PUBLICATION supabase_realtime ADD TABLE staging.attendance_logs;
    EXCEPTION WHEN duplicate_object THEN NULL;
    END;
    BEGIN
      ALTER PUBLICATION supabase_realtime ADD TABLE staging.system_settings;
    EXCEPTION WHEN duplicate_object THEN NULL;
    END;
    BEGIN
      ALTER PUBLICATION supabase_realtime ADD TABLE staging.personnel;
    EXCEPTION WHEN duplicate_object THEN NULL;
    END;
    BEGIN
      ALTER PUBLICATION supabase_realtime ADD TABLE staging.holidays;
    EXCEPTION WHEN duplicate_object THEN NULL;
    END;
  END IF;
END $$;

NOTIFY pgrst, 'reload schema';

1.2 สรุปคำสั่ง SQL สะสมทั้งหมดที่ต้องนำไปใช้กับ public (อัปเดตใน sql/alter_v2_8_loca_v2_schema.sql)

รวบรวมทุกการเปลี่ยนแปลงโครงสร้างฐานข้อมูลที่เกิดขึ้นใน staging (ตั้งแต่ V2.8 Loca-V2 จนถึงปัจจุบัน) เพื่อให้นำไปรันบน public ได้ครบจบในชุดเดียวเมื่อรวมเข้า main:

-- ============================================================================
-- [PUBLIC CUMULATIVE MIGRATION] สรุปคำสั่ง SQL สะสมสำหรับนำไปใช้กับ Schema public
-- ไฟล์: sql/alter_v2_8_loca_v2_schema.sql
-- ============================================================================

-- 1. ตาราง public.attendance_logs: รองรับโหมดในสถานที่/นอกสถานที่ และเหตุผลช่วยราชการ
ALTER TABLE public.attendance_logs
  ADD COLUMN IF NOT EXISTS work_mode TEXT DEFAULT 'onsite',
  ADD COLUMN IF NOT EXISTS offsite_reason TEXT DEFAULT NULL,
  ADD COLUMN IF NOT EXISTS check_out_work_mode TEXT DEFAULT NULL,
  ADD COLUMN IF NOT EXISTS check_out_offsite_reason TEXT DEFAULT NULL;

-- 2. ตาราง public.system_settings: รองรับสวิตช์บังคับตรวจสอบรัศมีสถานที่ทำงาน (Geofence)
ALTER TABLE public.system_settings
  ADD COLUMN IF NOT EXISTS enforce_geofence_radius BOOLEAN NOT NULL DEFAULT true;

UPDATE public.system_settings
SET enforce_geofence_radius = COALESCE(
  (category_colors->>'enforce_geofence_radius')::boolean,
  enforce_geofence_radius,
  true
);

-- 3. เปิดใช้งาน Supabase Realtime WebSocket สำหรับตารางในสคีมา public
DO $$
BEGIN
  IF EXISTS (SELECT 1 FROM pg_publication WHERE pubname = 'supabase_realtime') THEN
    BEGIN
      ALTER PUBLICATION supabase_realtime ADD TABLE public.attendance_logs;
    EXCEPTION WHEN duplicate_object THEN NULL;
    END;
    BEGIN
      ALTER PUBLICATION supabase_realtime ADD TABLE public.system_settings;
    EXCEPTION WHEN duplicate_object THEN NULL;
    END;
    BEGIN
      ALTER PUBLICATION supabase_realtime ADD TABLE public.personnel;
    EXCEPTION WHEN duplicate_object THEN NULL;
    END;
    BEGIN
      ALTER PUBLICATION supabase_realtime ADD TABLE public.holidays;
    EXCEPTION WHEN duplicate_object THEN NULL;
    END;
  END IF;
END $$;
